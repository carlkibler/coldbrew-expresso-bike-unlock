# HID key script: bootstrap WiFi on the bike and report IP back to mule
# Placeholders {SSID}, {PASS}, {MULE_IP} are substituted by the orchestrator at runtime.

TYPE ip link show wlan0 >/dev/null 2>&1 && echo WIFI || echo NOWIFI | nc -q0 {MULE_IP} 9998
ENTER
WAIT 2000
TYPE sudo service network-manager stop 2>/dev/null; sudo ifconfig wlan0 up
ENTER
WAIT 1000
TYPE wpa_passphrase {SSID} {PASS} | sudo tee /etc/wpa_supplicant/coldbrew.conf >/dev/null
ENTER
WAIT 500
TYPE sudo wpa_supplicant -B -i wlan0 -c /etc/wpa_supplicant/coldbrew.conf
ENTER
WAIT 5000
TYPE sudo dhclient wlan0
ENTER
WAIT 4000
TYPE ip -4 addr show wlan0 | nc -q0 {MULE_IP} 9999
ENTER
